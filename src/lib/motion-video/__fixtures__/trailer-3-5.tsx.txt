// --- BEAT 1: HOOK ---
// wow: FULL_CANVAS_SCALE — "You need five seconds." expands and pushes into the tension scene.
const HookBeat: React.FC = () => {
	const frame = useCurrentFrame();
	const { fps: f } = useVideoConfig();

	// Starting word opacity is initialized at 1 / starts immediately from frame 0 for first word
	const drift = interpolate(frame, [0, 92], [10, -18], { easing: EXPO, ...CLAMP });
	const blow = interpolate(frame, [72, 92], [1, 2.8], { easing: EXPO, ...CLAMP });
	const blowFade = interpolate(frame, [78, 92], [1, 0], { easing: EXPO, ...CLAMP });
	const line2Show = interpolate(frame, [1.1 * f, 1.7 * f], [0, 1], { easing: EXPO, ...CLAMP });
	const line2Y = interpolate(frame, [1.1 * f, 1.7 * f], [36, 0], { easing: SETTLE, ...CLAMP });
	const logoOp = interpolate(frame, [0, 10], [0.5, 0.95], { easing: EXPO, ...CLAMP });

	const words = ["You", "don't", "need", "a", "marketing", "department."];

	return (
		<AbsoluteFill style={{ backgroundColor: C.ink, opacity: blowFade, overflow: 'hidden' }}>
			<LogoMark opacity={logoOp} />
			<div
				style={{
					position: 'absolute',
					left: 72,
					right: 72,
					top: 620,
					transform: 'translateY(' + drift + 'px) scale(' + blow + ')',
					transformOrigin: '40% 65%',
				}}
			>
				<div style={{ display: 'flex', flexWrap: 'wrap', maxWidth: 940 }}>
					{words.map((w, i) => {
						const delay = i * 4;
						const rise = interpolate(frame - delay, [0, 12], [i === 0 ? 0 : 32, 0], {
							easing: EXPO,
							...CLAMP,
						});
						const show = interpolate(frame - delay, [0, 8], [i === 0 ? 1 : 0, 1], {
							easing: EXPO,
							...CLAMP,
						});
						return (
							<span
								key={w}
								style={{
									display: 'inline-block',
									marginRight: 16,
									opacity: show,
									transform: 'translateY(' + rise + 'px)',
									color: C.paper,
									fontFamily: 'Inter, sans-serif',
									fontSize: 64,
									fontWeight: 300,
									lineHeight: 1.2,
									letterSpacing: -1.2,
								}}
							>
								{w}
							</span>

					})}
				</div>

					style={{
						marginTop: 36,

};

// --- MAIN SCENE WRAPPER WITH SLIDE INERTIA & PARALLAX ---
// wow: SLIDE_INERTIA — UI scenes push laterally with inertia and continuous movement through cuts
export default function AnomaliaProductTrailer() {
	const frame = useCurrentFrame();

	// Timeline markers (in absolute frames)
	// 0 - 92: Hook
	// 82 - 172: Old Way
	// 162 - 270: Overview UI
	// 256 - 364: Calendar UI
	// 350 - 450: Editorial Plan UI
	// 436 - 540: CTA End Card

	// Scene 1: Hook (0..92)
	const s1Active = frame < 92;

	// Scene 2: Old Way (82..172)
	const s2Active = frame >= 82 && frame < 172;
	const s2Local = frame - 82;

	// Scene 3: Overview UI (162..270)
	const s3Active = frame >= 162 && frame < 270;
	const s3Local = frame - 162;
	// Lateral push exit for Scene 3
	const s3ExitX = interpolate(frame, [256, 270], [0, -width], { easing: EXPO, ...CLAMP });

	// Scene 4: Calendar UI (256..364)
	const s4Active = frame >= 256 && frame < 364;
	const s4Local = frame - 256;
	// Lateral push enter for Scene 4 (with inertia overshoot)
	const s4EnterX = interpolate(frame, [256, 270], [width, -18], { easing: EXPO, ...CLAMP });
	const s4SettleX = interpolate(frame, [270, 280], [0, 18], { easing: SETTLE, ...CLAMP });
	const s4ExitX = interpolate(frame, [350, 364], [0, -width], { easing: EXPO, ...CLAMP });
	const s4PosX = s4EnterX + s4SettleX + s4ExitX;

	// Scene 5: Editorial Plan UI (350..450)
	const s5Active = frame >= 350 && frame < 450;
	const s5Local = frame - 350;
	// Lateral push enter for Scene 5
	const s5EnterX = interpolate(frame, [350, 364], [width, -18], { easing: EXPO, ...CLAMP });
	const s5SettleX = interpolate(frame, [364, 374], [0, 18], { easing: SETTLE, ...CLAMP });
	const s5ExitScale = interpolate(frame, [436, 450], [1, 0.88], { easing: EXPO, ...CLAMP });
	const s5ExitFade = interpolate(frame, [436, 450], [1, 0], { easing: EXPO, ...CLAMP });
	const s5PosX = s5EnterX + s5SettleX;

	// Scene 6: CTA End Card (436..540) — Direct fluid scale reveal without black gap!
	const s6Active = frame >= 436;
	const s6Local = frame - 436;

	return (
		<AbsoluteFill style={{ backgroundColor: C.ink }}>
			{/* Beat 1: Hook */}
			{s1Active && <HookBeat />}

			{/* Beat 2: Old Way */}
			{s2Active && (
				<AbsoluteFill style={{ zIndex: 5 }}>
					<OldWayBeat />
				</AbsoluteFill>
			)}

			{/* Beat 3: Overview UI */}
			{s3Active && (
				<AbsoluteFill
					style={{
						zIndex: 10,
						transform: 'translateX(' + s3ExitX + 'px)',
					}}
				>
					<OverviewBeat len={108} />
				</AbsoluteFill>
			)}

			{/* Beat 4: Calendar UI */}
			{s4Active && (
				<AbsoluteFill
					style={{
						zIndex: 15,
						transform: 'translateX(' + s4PosX + 'px)',
					}}
				>
					<CalendarBeat len={108} />
				</AbsoluteFill>
			)}

			{/* Beat 5: Editorial Plan UI */}
			{s5Active && (
				<AbsoluteFill
					style={{
						zIndex: 20,
						transform: 'translateX(' + s5PosX + 'px) scale(' + s5ExitScale + ')',
						opacity: s5ExitFade,
					}}
				>
					<PlanBeat len={100} />
				</AbsoluteFill>
			)}

			{/* Beat 6: Resolution CTA End Card */}
			{s6Active && (
				<AbsoluteFill style={{ zIndex: 25 }}>
					<CtaEndCard len={104} />
				</AbsoluteFill>
			)}
		</AbsoluteFill>
	);
}
